SQL Database Schema Definition (.sql)

-- MarketLink Database Schema
CREATE TABLE users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    role VARCHAR(20) NOT NULL -- 'customer' or 'farmer'
);

CREATE TABLE products (
    product_id INT PRIMARY KEY AUTO_INCREMENT,
    farmer_id INT,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50),
    price DECIMAL(10,2) NOT NULL,
    stock_quantity INT NOT NULL,
    status VARCHAR(20) -- 'Available' or 'Sold Out'
);

CREATE TABLE orders (
    order_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_name VARCHAR(100),
    items_summary TEXT,
    total_amount DECIMAL(10,2),
    pickup_slot VARCHAR(50),
    order_status VARCHAR(30) -- 'Placed', 'Accepted', 'Ready for Pickup'
);
